/** The accepted campaign is exactly eight paired rounds with both public operations. */
import test from 'node:test';
import assert from 'node:assert/strict';
import { execFileSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
test('fixed paired plan preserves all 32 fresh workers and the accepted sample counts', () => {
  const jobs = JSON.parse(execFileSync(process.execPath, [fileURLToPath(new URL('./campaign.mjs', import.meta.url)), '--plan'], { encoding: 'utf8' }));
  assert.equal(jobs.length, 32);
  for (let round = 0; round < 8; round++) {
    const observed = jobs.filter(job => job.round === round);
    const arms = round % 2 ? ['canonical-candidate', 'accepted-main'] : ['accepted-main', 'canonical-candidate'];
    assert.deepEqual(observed.map(job => [job.phase, job.artifact]), ['parse', 'parse-with-positions'].flatMap(phase => arms.map(arm => [phase, arm])));
    assert.ok(observed.every(job => job.warmup === 5 && job.iterations === 30 && job.mode === 'timing'));
  }
});
